
		private void Level%%LEVEL%%()
		{
			_builder
				.AddLevel()
				.HP(%%HP%%)
				.FP(%%FP%%)
				.STM(%%STM%%)
				.DMG(%%DMG%%)
				.Delay(%%DELAY%%)
				.Stat(AbilityType.Might, %%MGT%%)
				.Stat(AbilityType.Perception, %%PER%%)
				.Stat(AbilityType.Vitality, %%VIT%%)
				.Stat(AbilityType.Willpower, %%WIL%%)
				.Stat(AbilityType.Agility, %%AGI%%)
				.Stat(AbilityType.Social, %%SOC%%)
				.MaxAttackBonus(%%MAXATTACKBONUS%%)
				.MaxAccuracyBonus(%%MAXACCURACYBONUS%%)
				.MaxEvasionBonus(%%MAXEVASIONBONUS%%)
				.MaxDefenseBonus(CombatDamageType.Physical, %%MAXPHYSICALDEFENSE%%)
				.MaxDefenseBonus(CombatDamageType.Force, %%MAXFORCEDEFENSE%%)
				.MaxResistanceBonus(ResistanceType.Fire, %%MAXFIREDEFENSE%%)
				.MaxResistanceBonus(ResistanceType.Poison, %%MAXPOISONDEFENSE%%)
				.MaxResistanceBonus(ResistanceType.Electrical, %%MAXELECTRICALDEFENSE%%)
				.MaxResistanceBonus(ResistanceType.Ice, %%MAXICEDEFENSE%%)
				.MaxResistanceBonus(ResistanceType.Mind, %%MAXWILL%%)
				.MaxResistanceBonus(ResistanceType.Trauma, %%MAXFORTITUDE%%)
				.MaxResistanceBonus(ResistanceType.Mobility, %%MAXREFLEX%%);
		}
